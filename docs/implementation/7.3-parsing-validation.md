# 7.3 Parsing et validation RFC

## Le cœur logiciel du MULS

Le **parsing et la validation** constituent le système nerveux central de toute implémentation MULS. Ces composants transforment les fichiers texte MULS en structures de données utilisables, tout en garantissant leur **conformité parfaite** aux spécifications RFC-0001.

Cette section détaille l'architecture, les algorithmes et les meilleures pratiques pour implémenter des parseurs et validateurs MULS robustes.

## Architecture générale

### Pipeline de traitement

```
Source MULS ──→ Lexer ──→ Parser ──→ AST ──→ Validator ──→ Structure validée
     │              │         │         │         │
     └─ Erreurs ────┴─────────┴─────────┴─────────┴─→ Rapport d'erreurs
```

### Composants principaux

**Lexer (Analyseur lexical)** : Transformation texte → tokens
**Parser (Analyseur syntaxique)** : Tokens → AST (Abstract Syntax Tree)
**Validator (Validateur sémantique)** : AST → structure validée
**Error Reporter** : Collecte et formatage des erreurs

### Implémentation du lexer

```python
class MULSLexer:
    def __init__(self, source):
        self.source = source
        self.position = 0
        self.line = 1
        self.column = 1
        self.tokens = []

    def tokenize(self):
        while self.position < len(self.source):
            if self.source[self.position].isspace():
                self.skip_whitespace()
            elif self.source[self.position] == '/':
                self.handle_comment()
            elif self.source[self.position] == '"':
                self.tokenize_string()
            elif self.source[self.position].isdigit() or self.source[self.position] == '-':
                self.tokenize_number()
            elif self.source[self.position].isalpha() or self.source[self.position] == '_':
                self.tokenize_identifier()
            else:
                self.tokenize_symbol()

        self.tokens.append(MULSToken('EOF', '', self.line, self.column))
        return self.tokens

    def skip_whitespace(self):
        while self.position < len(self.source) and self.source[self.position].isspace():
            if self.source[self.position] == '\n':
                self.line += 1
                self.column = 1
            else:
                self.column += 1
            self.position += 1

    def tokenize_identifier(self):
        start = self.position
        while (self.position < len(self.source) and
               (self.source[self.position].isalnum() or self.source[self.position] == '_')):
            self.position += 1
            self.column += 1

        value = self.source[start:self.position]
        token_type = self.get_keyword_type(value) if self.is_keyword(value) else 'IDENTIFIER'
        self.tokens.append(MULSToken(token_type, value, self.line, self.column - len(value)))
```

## Analyseur syntaxique (Parser)

### Grammaire formelle

```
universe ::= "UNIVERSE" "{" signature morphs "}"

signature ::= "SIG" "(" NUMBER "," NUMBER "," NUMBER "," NUMBER "," NUMBER ")"

morphs ::= morph*

morph ::= MORPH_TYPE ":" STRING ":" relations

relations ::= "(" relation_list ")" | "(" ")"

relation_list ::= relation ("," relation)*

relation ::= IDENTIFIER | IDENTIFIER "(" parameters ")"

parameters ::= parameter ("," parameter)*

parameter ::= IDENTIFIER "=" value

value ::= NUMBER | STRING | IDENTIFIER | "true" | "false"
```

### Implémentation recursive descent

```python
class MULSParser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.position = 0
        self.errors = []

    def parse(self):
        try:
            universe = self.parse_universe()
            if self.current_token().type != 'EOF':
                self.error("Tokens supplémentaires après la fin de l'univers")
            return universe
        except ParseError as e:
            self.errors.append(str(e))
            return None

    def parse_universe(self):
        self.expect('UNIVERSE')
        self.expect('LBRACE')

        signature = self.parse_signature()
        morphs = []

        while self.current_token().type in ['S', 'B', 'C', 'E', 'X', 'T']:
            morphs.append(self.parse_morph())

        self.expect('RBRACE')

        return MULSUniverse(signature, morphs)

    def parse_signature(self):
        self.expect('SIG')
        self.expect('LPAREN')

        ns = self.parse_number()
        self.expect('COMMA')
        nb = self.parse_number()
        self.expect('COMMA')
        nc = self.parse_number()
        self.expect('COMMA')
        deff = self.parse_number()
        self.expect('COMMA')
        chi = self.parse_number()

        self.expect('RPAREN')

        return MULSSignature(ns, nb, nc, deff, chi)

    def parse_morph(self):
        morph_type = self.current_token().type
        self.advance()

        self.expect('COLON')

        name = self.parse_string()

        self.expect('COLON')

        relations = self.parse_relations()

        return MULSMorph(morph_type, name, relations)
```

### Gestion des erreurs

```python
class ParseError(Exception):
    def __init__(self, message, token=None):
        self.message = message
        self.token = token
        self.line = token.line if token else 0
        self.column = token.column if token else 0

class MULSErrorReporter:
    def __init__(self):
        self.errors = []
        self.warnings = []

    def error(self, message, token=None):
        self.errors.append(ParseError(message, token))

    def warning(self, message, token=None):
        self.warnings.append(ParseError(message, token))

    def report(self):
        for error in self.errors:
            print(f"Erreur ligne {error.line}, colonne {error.column}: {error.message}")
        for warning in self.warnings:
            print(f"Avertissement ligne {warning.line}, colonne {warning.column}: {warning.message}")
```

## Validateur sémantique

### Types de validation

**Syntaxique** : Grammaire respectée
**Sémantique** : Signification physique cohérente
**Structurelle** : Intégrité topologique
**Physique** : Compatibilité avec les lois connues

### Implémentation du validateur

```python
class MULSValidator:
    def __init__(self, universe):
        self.universe = universe
        self.errors = []
        self.warnings = []

    def validate(self):
        self.validate_signature()
        self.validate_morphs()
        self.validate_relations()
        self.validate_topology()
        self.validate_physics()

        return {
            'valid': len(self.errors) == 0,
            'errors': self.errors,
            'warnings': self.warnings
        }

    def validate_signature(self):
        sig = self.universe.signature

        # Vérifications de base
        if sig.ns < 0 or sig.nb < 0 or sig.nc < 0:
            self.error("Les composantes de signature ne peuvent être négatives")

        if sig.deff < 4:
            self.error("Les dimensions effectives doivent être ≥ 4")

        # Vérification de cohérence
        actual_ns = len([m for m in self.universe.morphs if m.type == 'S'])
        if actual_ns != sig.ns:
            self.error(f"Nombre de cordes: attendu {sig.ns}, trouvé {actual_ns}")

    def validate_morphs(self):
        morph_ids = set()

        for morph in self.universe.morphs:
            # Unicité des IDs
            if morph.id in morph_ids:
                self.error(f"ID de morph dupliqué: {morph.id}")
            morph_ids.add(morph.id)

            # Validation des propriétés
            self.validate_morph_properties(morph)

    def validate_relations(self):
        morph_ids = {m.id for m in self.universe.morphs}

        for morph in self.universe.morphs:
            for relation in morph.relations:
                # Vérifier que la cible existe
                if relation.target not in morph_ids:
                    self.error(f"Relation vers morph inexistant: {relation.target}")

                # Valider le type de relation
                self.validate_relation_type(morph.type, relation)

    def validate_topology(self):
        # Calcul de la caractéristique d'Euler
        calculated_chi = self.calculate_euler_characteristic()
        if calculated_chi != self.universe.signature.chi:
            self.error(f"Caractéristique d'Euler: attendu {self.universe.signature.chi}, calculé {calculated_chi}")

    def validate_physics(self):
        # Vérifications physiques de base
        for morph in self.universe.morphs:
            if morph.type == 'S' and 'mass' in morph.properties:
                if morph.properties['mass'] < 0:
                    self.error(f"Masse négative pour {morph.id}")
```

### Mode de récupération

```python
class ErrorRecoveryParser(MULSParser):
    def __init__(self, tokens):
        super().__init__(tokens)
        self.recovery_points = []

    def recover_from_error(self, error_token):
        # Chercher un point de récupération
        while self.position < len(self.tokens):
            if self.is_recovery_point(self.current_token()):
                break
            self.advance()

        # Marquer l'erreur comme récupérée
        self.errors.append(f"Récupération d'erreur près de {error_token.value}")
```
