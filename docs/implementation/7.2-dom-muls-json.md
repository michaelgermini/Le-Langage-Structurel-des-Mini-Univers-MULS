# 7.2 DOM-MULS en JSON

## L'interface programmable des univers MULS

Le **DOM-MULS (Document Object Model for MULS)** est une représentation JSON standardisée qui transforme les univers MULS en **objets JavaScript natifs**. Cette approche permet l'intégration transparente du MULS dans les applications web, les APIs REST, et les écosystèmes de programmation moderne.

Le DOM-MULS n'est pas seulement une sérialisation ; c'est une **interface de programmation complète** avec méthodes, événements et manipulation dynamique.

## API JavaScript

### Classe principale MULSUniverse

```javascript
class MULSUniverse {
  constructor(jsonData = null) {
    this.version = "RFC-0001";
    this.id = null;
    this.signature = null;
    this.metadata = {};
    this.morphs = new Map();

    if (jsonData) {
      this.loadFromJSON(jsonData);
    }
  }

  // Chargement depuis JSON
  loadFromJSON(jsonData) {
    this.validateJSON(jsonData);

    const universe = jsonData.universe;
    this.id = universe.id;
    this.signature = universe.signature;
    Object.assign(this.metadata, universe.metadata);

    // Charger les morphs
    for (const [id, morphData] of Object.entries(universe.morphs)) {
      this.morphs.set(id, new MULSMorph(id, morphData));
    }

    this.resolveRelations();
  }

  // Validation du JSON
  validateJSON(jsonData) {
    // Utiliser JSON Schema validator
    const isValid = this.validateAgainstSchema(jsonData);
    if (!isValid) {
      throw new Error("JSON invalide selon le schéma MULS");
    }
  }

  // Résolution des relations croisées
  resolveRelations() {
    for (const morph of this.morphs.values()) {
      morph.resolveRelations(this.morphs);
    }
  }

  // Export vers JSON
  toJSON() {
    return {
      muls_version: this.version,
      universe: {
        id: this.id,
        signature: this.signature,
        metadata: this.metadata,
        morphs: Object.fromEntries(
          Array.from(this.morphs.entries()).map(([id, morph]) => [
            id, morph.toJSON()
          ])
        )
      }
    };
  }
}
```

### Classe MULSMorph

```javascript
class MULSMorph {
  constructor(id, data) {
    this.id = id;
    this.type = data.type;
    this.properties = data.properties || {};
    this.relations = new Map();
    this.relatedMorphs = new Map(); // Résolus

    // Charger les relations non résolues
    if (data.relations) {
      for (const [targetId, relationData] of Object.entries(data.relations)) {
        this.relations.set(targetId, relationData);
      }
    }
  }

  // Résoudre les références
  resolveRelations(allMorphs) {
    for (const [targetId, relationData] of this.relations) {
      if (allMorphs.has(targetId)) {
        this.relatedMorphs.set(targetId, {
          morph: allMorphs.get(targetId),
          relation: relationData
        });
      } else {
        console.warn(`Morph ${targetId} référencé mais non trouvé`);
      }
    }
  }

  // Ajouter une relation
  addRelation(targetId, relationData) {
    this.relations.set(targetId, relationData);
    // Re-résoudre si nécessaire
  }

  // Calculer des propriétés dérivées
  getDegree() {
    return this.relations.size;
  }

  getCentrality() {
    // Calcul de centralité locale
    return this.calculateCentrality();
  }

  toJSON() {
    return {
      type: this.type,
      properties: this.properties,
      relations: Object.fromEntries(this.relations)
    };
  }
}
```

## Méthodes de manipulation

### Création et modification

```javascript
// Créer un univers
const universe = new MULSUniverse();

// Ajouter un morph
universe.addMorph("electron", {
  type: "S",
  properties: { mass: 0.511, charge: -1 },
  relations: {
    "brane1": { type: "confinement" }
  }
});

// Modifier un morph
const electron = universe.getMorph("electron");
electron.properties.spin = 0.5;

// Ajouter une relation
electron.addRelation("compact1", {
  type: "vibration",
  strength: 0.8
});
```

## Visualisation interactive

### Avec D3.js

```javascript
function createMULSVisualization(universe) {
  const svg = d3.select("#muls-canvas");

  // Créer les nœuds
  const nodes = universe.morphs.map(morph => ({
    id: morph.id,
    type: morph.type,
    x: Math.random() * 800,
    y: Math.random() * 600
  }));

  // Créer les liens
  const links = [];
  universe.morphs.forEach(morph => {
    morph.relatedMorphs.forEach(({ morph: target, relation }) => {
      links.push({
        source: morph.id,
        target: target.id,
        type: relation.type,
        strength: relation.strength || 1
      });
    });
  });

  // Simulation de force
  const simulation = d3.forceSimulation(nodes)
    .force("link", d3.forceLink(links).id(d => d.id))
    .force("charge", d3.forceManyBody())
    .force("center", d3.forceCenter(400, 300));

  // Dessiner les liens
  const link = svg.selectAll("line")
    .data(links)
    .enter().append("line")
    .attr("stroke", d => getColorForRelation(d.type));

  // Dessiner les nœuds
  const node = svg.selectAll("circle")
    .data(nodes)
    .enter().append("circle")
    .attr("r", 10)
    .attr("fill", d => getColorForType(d.type))
    .call(d3.drag()
      .on("start", dragstarted)
      .on("drag", dragged)
      .on("end", dragended));

  // Mise à jour des positions
  simulation.on("tick", () => {
    link.attr("x1", d => d.source.x)
        .attr("y1", d => d.source.y)
        .attr("x2", d => d.target.x)
        .attr("y2", d => d => d.target.y);

    node.attr("cx", d => d.x)
        .attr("cy", d => d.y);
  });
}
```

## Performance et optimisation

### Lazy loading

```javascript
class LazyMULSUniverse extends MULSUniverse {
  constructor() {
    super();
    this.loadedMorphs = new Set();
  }

  async getMorph(id) {
    if (!this.loadedMorphs.has(id)) {
      await this.loadMorph(id);
      this.loadedMorphs.add(id);
    }
    return this.morphs.get(id);
  }

  async loadMorph(id) {
    const response = await fetch(`/api/morphs/${id}`);
    const morphData = await response.json();
    this.morphs.set(id, new MULSMorph(id, morphData));
  }
}
```

### Cache et indexation

```javascript
class CachedMULSUniverse extends MULSUniverse {
  constructor() {
    super();
    this.cache = new Map();
    this.indices = {
      byType: new Map(),
      byProperty: new Map()
    };
  }

  addMorph(id, morphData) {
    super.addMorph(id, morphData);

    // Mettre à jour les index
    const morph = this.morphs.get(id);
    if (!this.indices.byType.has(morph.type)) {
      this.indices.byType.set(morph.type, new Set());
    }
    this.indices.byType.get(morph.type).add(id);
  }

  findByType(type) {
    return Array.from(this.indices.byType.get(type) || []);
  }
}
```

function validateMULSSecurity(jsonData) {
  // Vérifier la taille
  const size = JSON.stringify(jsonData).length;
  if (size > MAX_UNIVERSE_SIZE) {
    throw new Error("Univers trop volumineux");

  // Vérifier la profondeur
  const depth = getJSONDepth(jsonData);

  if (hasCircularReferences(jsonData)) {

}
